คุณสามารถเชื่อมต่อข้อมูลเข้ากับหน้า Dashboard นี้ได้ 2 วิธี: (1) วาง Web App URL จาก Google Apps Script เพื่อดึงข้อมูลออนไลน์อัตโนมัติ หรือ (2) อัปโหลดไฟล์ CSV (`PaobeiPao_Master_PL_Transactions.csv`) จากเครื่องของคุณเพื่อวิเคราะห์ทันที

วิธีที่ 1: เชื่อมต่อผ่าน Google Apps Script Web App URL (Real-time)

วิธีที่ 2: อัปโหลดไฟล์ CSV จาก Google Sheets ของคุณโดยตรง

📋 โค้ด Google Apps Script สำหรับใส่ในไฟล์ PaobeiPao_Master_PL

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var txSheet = ss.getSheetByName("Transactions");
  var evSheet = ss.getSheetByName("Event_Master");
  
  var txRows = txSheet ? txSheet.getDataRange().getValues() : [];
  var evRows = evSheet ? evSheet.getDataRange().getValues() : [];
  
  var transactions = [];
  for (var i = 1; i < txRows.length; i++) {
    var r = txRows[i];
    if (!r[0] || !r[7]) continue;
    var d = r[0] instanceof Date ? Utilities.formatDate(r[0], "GMT+7", "yyyy-MM-dd") : String(r[0]).slice(0,10);
    transactions.push({
      date: d,
      brand: String(r[1] || "Baobei").trim(),
      model: String(r[2] || "สาขา (Branch)").trim(),
      subLoc: String(r[3] || "").trim(),
      type: String(r[4] || "").trim(),      // Revenue / Expense
      costType: String(r[5] || "").trim(),  // COGS / Direct Expense / Fixed Overhead
      category: String(r[6] || "").trim(),
      amount: Number(r[7]) || 0,
      note: String(r[8] || "").trim(),
      periodKey: String(r[9] || d.slice(0,7)).trim()
    });
  }
  
  var events = [];
  for (var j = 1; j < evRows.length; j++) {
    var er = evRows[j];
    if (!er[0]) continue;
    events.push({
      id: String(er[0]), name: String(er[1]), brand: String(er[2]),
      location: String(er[3]),
      start: er[4] instanceof Date ? Utilities.formatDate(er[4], "GMT+7", "yyyy-MM-dd") : String(er[4]),
      end: er[5] instanceof Date ? Utilities.formatDate(er[5], "GMT+7", "yyyy-MM-dd") : String(er[5]),
      gpPct: typeof er[6] === 'number' ? (er[6] <= 1 ? er[6]*100 : er[6]) : parseFloat(String(er[6]).replace('%','')) || 0,
      fixedRent: Number(er[7]) || 0,
      status: String(er[8] || "Completed")
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "ok", transactions: transactions, events: events
  })).setMimeType(ContentService.MimeType.JSON);
}
